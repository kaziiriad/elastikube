# Production dependencies for the ml-training-cronjob pipeline.
#
# Pipeline scripts and their direct deps:
#   scripts/extract_data.py  -> boto3, pandas
#   scripts/train_model.py   -> boto3, pandas, prophet, joblib
#   scripts/validate_model.py -> pandas, prophet
#   utils/feature_engineer.py -> pandas, numpy
#
# prophet is installed with --no-deps in the Dockerfile so matplotlib /
# pillow / kiwisolver / fontTools / contourpy / mpl_toolkits are NOT
# pulled in. Prophet's __init__ emits a benign "Importing matplotlib
# failed" warning at import time, but the forecasting pipeline
# (fit, predict, serialize, cross_validation) does not need matplotlib.
# Plotting libraries are dev-only and live in requirements-dev.txt.
#
# The Dockerfile installs:
#   1. requirements-deps.txt   -> boto3, pandas, numpy, joblib, cmdstanpy,
#                                  tqdm, importlib_resources, holidays
#   2. prophet                 -> --no-deps (matplotlib skipped)
#   3. the local package       -> so scripts can `from utils...`
#
# Pinned loosely to keep the image reproducible. Bump as upstream
# releases ship.
boto3>=1.34.0,<2.0.0
pandas>=2.0.0,<3.0.0
numpy>=1.24.0,<3.0.0
joblib>=1.3.0,<2.0.0

# Prophet runtime deps. Listed explicitly because the Dockerfile
# installs them with full dependency resolution (from
# requirements-deps.txt), then installs prophet last with --no-deps.
cmdstanpy>=1.0.4
tqdm>=4.36.1
importlib_resources
holidays>=0.25,<1

# Prophet itself (installed --no-deps from the Dockerfile).
prophet>=1.1.4,<2.0.0
